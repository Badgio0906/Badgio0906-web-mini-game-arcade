function render(): void {

  const canDrop = playScreen() && board.result === 'playing' && humanTurn;
  const focused = document.activeElement instanceof HTMLElement && document.activeElement.closest('#board');
  const ghost = canDrop ? board.landing(selectedColumn) : null;

  boardElement.innerHTML = Array.from({ length: COLS }, (_, column) => {
    const full = board.landing(column) === null;
    const contents = Array.from({ length: ROWS }, (_, row) => {
      const index = row * COLS + column, value = board.cells[index];
      const preview = index === ghost && !value;
      return `<span aria-hidden="true" class="slot ${value === 1 ? 'green' : value === 2 ? 'amber' : ''} ${preview ? `ghost ${board.turn === 2 ? 'amber-ghost' : ''}` : ''} ${board.wins.includes(index) ? 'winner' : ''} ${index === lastIndex ? 'last' : ''}">${value === 1 || preview && board.turn === 1 ? '●' : value === 2 || preview && board.turn === 2 ? '◆' : ''}</span>`;
    }).join('');
    const stack = Array.from({ length: ROWS }, (_, row) => board.cells[row * COLS + column]).filter(Boolean).map(playerName).join('、');
    return `<button type="button" class="column ${column === selectedColumn ? 'selected' : ''} ${column === highlighted ? 'hinted' : ''}" data-game-interaction data-column="${column}" tabindex="${column === selectedColumn ? 0 : -1}" aria-label="${column + 1}列目${full ? '、満杯' : '、置ける'}${stack ? '。上から ' + stack : '。空の列'}" ${!canDrop || full ? 'disabled' : ''}><span class="col-number" aria-hidden="true">${column + 1}</span>${contents}</button>`;
  }).join('');
  if (focused && canDrop) focusColumn();
  el('turn-label').textContent = board.result !== 'playing' ? (board.result === 'draw' ? '引き分け' : `${playerName(board.result === 'green' ? 1 : 2)}の勝ち`) : training ? '練習：まるを4つ' : settings.mode === 'two' ? `${playerName(board.turn)}の番` : humanTurn ? 'あなたの番' : 'CPUの番';

  el('live-status').textContent = status;

function start(): void {
  cancelJob(); training = false;
  if (archive.saved && !archive.saved.reported) { active = true; settings = { ...archive.saved.settings }; board = FourBoard.fromMoves(archive.saved.moves)!; if (archive.saved.runId) telemetry.restoreRun(archive.saved.runId); end('reset'); settings = { ...archive.settings }; }
  board = new FourBoard(); highlighted = null; selectedColumn = 3;
  const seed = Math.floor(Math.random() * 0x100000000);
  archive.saved = { moves: [], settings: { ...settings }, runId: null, seed, hints: 0, undos: 0, reported: false, outcome: null };
  active = true; telemetry.trackEvent('run_start', details()); archive.saved.runId = telemetry.getActiveRunId(); save();
  status = settings.mode === 'two' ? '緑・まるから。列を選んで置こう。' : settings.first ? 'あなたは緑・まる。列を選んで置こう。' : 'あなたは琥珀・ひし形。CPUの初手を待とう。';

function resumeSaved(): void {
  const saved = archive.saved; if (!saved || saved.reported) return;
  cancelJob(); training = false; settings = { ...saved.settings }; board = FourBoard.fromMoves(saved.moves)!; highlighted = null; active = true;
  if (saved.runId) telemetry.restoreRun(saved.runId);
  telemetry.trackEvent('specific_game_events', { event: 'restore', ...details() });
  status = '保存した対局を再開しました。'; gate.resetInput(); screen('playing');
  if (board.result !== 'playing') result(); else scheduleCpu();
  focusColumn(); void audio.unlock();
}

function drop(column: number): void {
  if (!playScreen() || job === 'hint') return;
  if (training && column !== 3) { status = '練習では4列目を選ぼう。まるが横に4つつながります。'; render(); return; }
  if (!gate.humanDrop(board, column, settings)) return;
  selectedColumn = column; highlighted = null; action('move'); save(); audio.tone(380, 270, .07, 'sine', 0, .025);
  if (board.result !== 'playing') { result(); return; }
  status = settings.mode === 'two' ? `${playerName(board.turn)}の番。端末を渡して、ひとつ置こう。` : 'CPUが考えています。待ったで直前の手を戻せます。';
  render(); scheduleCpu();
}
function beginJob(kind: 'cpu' | 'hint'): void {
  cancelJob(); job = kind;
  const ticket = gate.ticket(board), moves = [...board.moves];
  const seed = (archive.saved?.seed ?? 1) ^ board.moves.length * 7919;
  const complete = (column: number | null) => {
    if (!playScreen() || !gate.valid(ticket, board) || job !== kind) return;
    worker?.terminate(); worker = null; if (fallbackTimer !== null) clearTimeout(fallbackTimer); fallbackTimer = null; job = null;
    if (column === null || !board.legalColumns().includes(column)) { status = '候補を見つけられませんでした。もう一度どうぞ。'; render(); return; }
    if (kind === 'hint') {
      highlighted = column; selectedColumn = column;
      if (archive.saved) archive.saved.hints++; action('hint'); save();
      status = `${column + 1}列目が候補です。自分で選んで置こう。`; render(); focusColumn();
    } else if (gate.cpuDrop(ticket, board, column, settings)) {
      highlighted = null; action('cpu_move'); save(); audio.tone(300, 230, .07, 'sine', 0, .02);
      status = `CPUは${column + 1}列目に置きました。あなたの番です。`;
      if (board.result !== 'playing') result(); else { render(); focusColumn(); }
    }
  };
  const fallback = async () => {
    if (!playScreen() || !gate.valid(ticket, board) || job !== kind) return;
    // An unavailable Worker gets a bounded tactical fallback, not unbounded UI search.
    const { chooseMove } = await import('./ai');
    if (!playScreen() || !gate.valid(ticket, board) || job !== kind) return;
    complete(chooseMove(FourBoard.fromMoves(moves)!, 'easy', { random: seededRandom(seed) }));
  };
  try {
    worker = new Worker(new URL('./ai.worker.ts', import.meta.url), { type: 'module' });
    worker.onmessage = event => {
      if (event.data.generation !== ticket.generation || event.data.revision !== ticket.revision) return;
      complete(event.data.column);
    };
    worker.onerror = event => { event.preventDefault(); worker?.terminate(); worker = null; void fallback(); };
    worker.postMessage({ ...ticket, moves, difficulty: kind === 'hint' ? 'strong' : settings.difficulty, seed });
    fallbackTimer = setTimeout(() => { worker?.terminate(); worker = null; void fallback(); }, 1800);
  } catch { fallbackTimer = setTimeout(() => { void fallback(); }, 100); }
  render();
}
