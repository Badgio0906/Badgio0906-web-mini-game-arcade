import{describe,it,expect}from'vitest';
import{createMatch,serve,award,step}from'../../src/games/game028/model';
import{MatchStore,freshSave,parseSave,finishReport,abandon,validMatch}from'../../src/games/game028/save';
import{MenuGesture,bindMenuPointerGuard}from'../../src/games/game028/input';
const running=()=>{const s=freshSave();s.active=true;s.reported=false;s.localReportId='local-test';s.match=createMatch();return s;};
describe('Game028 atomic save and inputs',()=>{
 it('round trips a live ball, all bounce phases, scores and nullable observer',()=>{const s=running();serve(s.match!);expect(parseSave(JSON.stringify(s))).toEqual(s);expect(s.observerRunId).toBeNull();});
 it('round trips naturally generated high uint32 RNG state without losing paused continuation',()=>{const s=running(),m=s.match!;serve(m);for(let n=0;n<360&&m.rally.status==='live'&&m.seed<=1e9;n++)step(m,1/120);expect(m.seed).toBeGreaterThan(1e9);expect(m.seed).toBeLessThanOrEqual(4294967295);expect(m.rally.status).toBe('live');expect(parseSave(JSON.stringify(s))).toEqual(s);});
 it('accepts RNG uint32 endpoints but rejects overflow, negative and fractional seeds',()=>{const s=running();for(const seed of[0,4294967295]){s.match!.seed=seed;expect(parseSave(JSON.stringify(s))).toEqual(s);}for(const seed of[4294967296,-1,1.5]){s.match!.seed=seed;expect(parseSave(JSON.stringify(s))).toEqual(freshSave());}s.match!.seed=28;s.match!.maxRally=1000000001;expect(validMatch(s.match!)).toBe(false);});
 it('rejects corrupted, practice and oversized snapshots',()=>{expect(parseSave('{')).toEqual(freshSave());const s=running();s.match!.practice=true;expect(parseSave(JSON.stringify(s))).toEqual(freshSave());expect(parseSave('x'.repeat(50001))).toEqual(freshSave());});
 it('rejects arbitrary analytics ID and impossible scores',()=>{const s=running();s.observerRunId='local-test';expect(parseSave(JSON.stringify(s))).toEqual(freshSave());s.observerRunId=null;s.match!.scores[0]=4;expect(parseSave(JSON.stringify(s))).toEqual(freshSave());});
 it('rejects NaN physics / mismatched CPU rally IDs',()=>{const m=createMatch();serve(m);m.rally.ball!.x=NaN;expect(validMatch(m)).toBe(false);m.rally.ball!.x=0;m.cpu.rallyId=0;expect(validMatch(m)).toBe(false);});
 it('failed writes permanently preserve latest memory without rereading stale valid backend',()=>{const old=freshSave();let reads=0;const store=new MatchStore({getItem:()=>{reads++;return JSON.stringify(old);},setItem:()=>{throw Error('denied');}});store.read();const s=running();store.write(s);expect(store.read()).toEqual(s);expect(reads).toBe(1);});
 it('semantic corruption retains current memory and rejects later stale backend reads',()=>{let raw=JSON.stringify(freshSave()),reads=0;const store=new MatchStore({getItem:()=>{reads++;return raw;},setItem:s=>{raw=s;}});const current=freshSave();current.stats.wins=7;store.write(current);raw=JSON.stringify({...current,version:99});expect(store.read().stats.wins).toBe(7);raw=JSON.stringify(freshSave());expect(store.read().stats.wins).toBe(7);expect(reads).toBe(1);});
 it('initial missing backend is legitimate fresh save, separate from malformed saved data',()=>{const store=new MatchStore({getItem:()=>null,setItem:()=>{}});expect(store.read()).toEqual(freshSave());});
 it('completed game with null observer reports local stats exactly once across restoration',()=>{const s=running(),m=s.match!;serve(m);m.mode=5;m.scores=[4,0];m.points=4;m.rally.status='live';award(m,0,'done');expect(finishReport(s,m)).toBe(true);expect(finishReport(s,m)).toBe(false);const restored=parseSave(JSON.stringify(s));expect(finishReport(restored,m)).toBe(false);expect(restored.stats.wins).toBe(1);expect(restored.stats.pointsFor).toBe(5);});
 it('rejects contradictory zero-score terminal and inactive live saves',()=>{const a=running();a.match!.rally.status='over';expect(parseSave(JSON.stringify(a))).toEqual(freshSave());const b=running();serve(b.match!);b.active=false;b.reported=true;expect(parseSave(JSON.stringify(b))).toEqual(freshSave());});
 it('rejects wrong winner or deuce-over without two-point gap',()=>{const s=running(),m=s.match!;serve(m);m.mode=11;m.scores=[11,10];m.points=21;m.rally.status='over';m.rally.winner=0;m.rally.scored=true;expect(validMatch(m)).toBe(false);m.scores=[12,10];m.points=22;m.rally.winner=1;expect(validMatch(m)).toBe(false);});
 it('accepts interrupted active over before report, then terminal ledger coherently once',()=>{const s=running(),m=s.match!;serve(m);m.mode=5;m.scores=[4,1];m.points=5;award(m,0,'done');expect(parseSave(JSON.stringify(s))).toEqual(s);const recovered=parseSave(JSON.stringify(s));expect(finishReport(recovered,recovered.match!)).toBe(true);expect(parseSave(JSON.stringify(recovered))).toEqual(recovered);expect(finishReport(recovered,recovered.match!)).toBe(false);});
 it('reported terminal needs the matching local ledger, not a completed-count equal ledger length',()=>{const s=running(),m=s.match!;serve(m);m.mode=5;m.scores=[4,0];m.points=4;award(m,0,'done');finishReport(s,m);s.stats.reportedIds.push('abandoned-earlier');expect(parseSave(JSON.stringify(s))).toEqual(s);s.stats.reportedIds=[];expect(parseSave(JSON.stringify(s))).toEqual(freshSave());});
 it('rejects scored ready or two-bounce live states',()=>{const m=createMatch();m.rally.scored=true;expect(validMatch(m)).toBe(false);m.rally.scored=false;serve(m);m.rally.ball!.bounces=2;expect(validMatch(m)).toBe(false);});
 it('abandon removes snapshot and cannot resurrect via pagehide continuation',()=>{const s=running();abandon(s,s.match!);expect(s.match).toBeNull();expect(s.active).toBe(false);expect(s.reported).toBe(true);expect(parseSave(JSON.stringify(s)).match).toBeNull();});
 it('capture menu guard rejects a preceding tap in another phase and allows fresh tap',()=>{const g=new MenuGesture(),b={}as Element;g.pointerDown(b);g.changePhase();expect(g.click(b,1)).toBe(false);g.pointerDown(b);expect(g.click(b,1)).toBe(true);expect(g.click(b,1)).toBe(false);});
 it('keyboard/assistive detail0 works but held activation cannot cross a phase',()=>{const g=new MenuGesture(),b={}as Element;expect(g.click(b,0)).toBe(true);expect(g.keyDown('Enter',false)).toBe(true);g.changePhase();expect(g.click(b,0)).toBe(false);expect(g.keyDown('Enter',true)).toBe(false);expect(g.keyUp('Enter')).toBe(false);expect(g.keyDown(' ',false)).toBe(true);expect(g.keyUp(' ')).toBe(true);});
 it('production capture listener sequence accepts native release once and rejects cancelled or phase-stale clicks',()=>{
 const menu=new EventTarget(),g=new MenuGesture(),button={}as Element;bindMenuPointerGuard(menu as HTMLElement,g);
 const emit=(name:string)=>{const e=new Event(name,{cancelable:true});Object.defineProperties(e,{target:{value:{closest:()=>button}},pointerId:{value:29},isPrimary:{value:true},button:{value:0},detail:{value:1}});menu.dispatchEvent(e);return !e.defaultPrevented;};
 emit('pointerdown');emit('pointerup');emit('lostpointercapture');expect(emit('click')).toBe(true);expect(emit('click')).toBe(false);
 emit('pointerdown');emit('pointerup');emit('pointercancel');emit('lostpointercapture');expect(emit('click')).toBe(false);
 emit('pointerdown');emit('pointerup');g.changePhase();emit('lostpointercapture');expect(emit('click')).toBe(false);
 emit('pointerdown');emit('lostpointercapture');expect(emit('click')).toBe(false);
 });
 it('native touch completed release survives implicit capture loss and activates once',()=>{const g=new MenuGesture(),b={}as Element;g.pointerDown(b,17);g.pointerUp(b,17);g.lostCapture(17);expect(g.click(b,1)).toBe(true);expect(g.click(b,1)).toBe(false);});
 it('unexpected capture loss, cancelled completed tap and old-phase release remain invalid',()=>{const b={}as Element;const a=new MenuGesture();a.pointerDown(b,1);a.lostCapture(1);expect(a.click(b,1)).toBe(false);const c=new MenuGesture();c.pointerDown(b,2);c.pointerUp(b,2);c.clearPointers();c.lostCapture(2);expect(c.click(b,1)).toBe(false);const d=new MenuGesture();d.pointerDown(b,3);d.pointerUp(b,3);d.changePhase();d.lostCapture(3);expect(d.click(b,1)).toBe(false);});
 it('different target pointerup and unrelated pointer cannot complete fresh menu press',()=>{const g=new MenuGesture(),b={}as Element,c={}as Element;g.pointerDown(b,1);g.pointerUp(b,2);g.lostCapture(1);expect(g.click(b,1)).toBe(false);g.pointerDown(b,3);g.pointerUp(c,3);expect(g.click(b,1)).toBe(false);});
 it('real cancelled pointer clears eligibility',()=>{const g=new MenuGesture(),b={}as Element;g.pointerDown(b);g.clearPointers();expect(g.click(b,1)).toBe(false);});
});
