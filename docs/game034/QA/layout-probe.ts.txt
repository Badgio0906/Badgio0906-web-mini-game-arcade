import {it} from 'vitest';import {writeFileSync} from 'node:fs';import {DemolitionRun,SITES} from '../../src/games/game034/DemolitionRun';
it('probes final-site physical content variants',()=>{const original=structuredClone(SITES[7].blocks),rows=[];for(const topX of [405,390,375])for(const midX of [365,375])for(const tankX of [445,420,400])for(const tankDensity of [.003,.0015]){const def=SITES[7];def.blocks=structuredClone(original);def.blocks.find(p=>p.id==='office-top')!.x=topX;def.blocks.find(p=>p.id==='office-mid')!.x=midX;def.blocks.find(p=>p.id==='office-tank')!.x=tankX;def.blocks.find(p=>p.id==='office-tank')!.density=tankDensity;let clean=0,best:any=null;const ids=def.blocks.filter(b=>b.cut).map(b=>b.id);for(let mask=1;mask<(1<<ids.length);mask++){const choices=ids.filter((_,i)=>mask&(1<<i));if(choices.length>3)continue;const r=new DemolitionRun(7);for(const id of choices)r.toggle(id);r.execute();for(let n=0;n<970&&r.phase==='simulating';n++)r.step(1/120);const s=r.result()!;if(s.clean){clean++;if(!best||s.score>best.score)best={...s,choices};}r.destroy();}rows.push({topX,midX,tankX,tankDensity,clean,best});}SITES[7].blocks=original;writeFileSync('docs/game034/QA/layout-variants.json',JSON.stringify(rows,null,2));},20000);
