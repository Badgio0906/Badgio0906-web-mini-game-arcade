import { length, type Point, type Obstacle } from './Geometry';
export type Level = { id: number; name: string; start: Point; socket: Point; obstacles: Obstacle[]; solution: Point[]; alternate?: Point[]; budget: number; hint: Point };
const p=(x:number,y:number):Point=>({x,y});
const o=(x:number,y:number,w:number,h:number,kind:Obstacle['kind']='desk'):Obstacle=>({x,y,w,h,kind});
function level(id:number,name:string,route:Point[],obstacles:Obstacle[],alternate?:Point[],extra=110):Level {
  return {id,name,start:route[0],socket:route.at(-1)!,solution:route,alternate,obstacles,budget:Math.ceil(Math.max(length(route),alternate?length(alternate):0)+extra),hint:route[1]};
}
