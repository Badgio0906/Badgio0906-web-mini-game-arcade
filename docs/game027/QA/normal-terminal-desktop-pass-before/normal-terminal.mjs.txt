import {chromium} from '@playwright/test';import fs from 'node:fs';import path from 'node:path';import {createHash} from 'node:crypto';
const base=process.env.GAME027_BASE||'http://127.0.0.1:4927',runId=new Date().toISOString().replace(/[:.]/g,'-'),hash=createHash('sha256').update(fs.readFileSync('src/games/game027/main.ts')).digest('hex').slice(0,12),dir=`docs/game027/QA/${hash}/${runId}-normal-terminal`;if(fs.existsSync(dir))throw Error('Refuse overwriting QA evidence');fs.mkdirSync(dir,{recursive:true});
const collectorHash=createHash('sha256').update(fs.readFileSync('tests/game027/normal-terminal.mjs')).digest('hex'),plannerHash=createHash('sha256').update(fs.readFileSync('tests/game027/geometry.mjs')).digest('hex');
const browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});let page,context,trace=[],shots=[];const errors=[];
const read=()=>page.evaluate(()=>JSON.parse(localStorage.getItem('web-mini-arcade:v1:game027:state')));
import {planShot} from './geometry.mjs';

async function aim(angle){const box=await page.locator('#table').boundingBox(),s=(await read()).snapshot,cue=s.world.balls[0];const x=box.x+(cue.x+50)/1000*box.width,y=box.y+(cue.y+50)/550*box.height;await page.mouse.move(x,y);await page.mouse.down();await page.mouse.move(x+Math.cos(angle)*80/1000*box.width,y+Math.sin(angle)*80/550*box.height);await page.mouse.up();}
async function power(n){const range=page.locator('#power');await range.focus();await page.keyboard.press('Home');for(let i=0;i<n;i++)await page.keyboard.press('ArrowRight');}
try{context=await browser.newContext({viewport:{width:1365,height:900}});page=await context.newPage();page.setDefaultTimeout(10000);page.on('pageerror',e=>errors.push(e.message));await context.route('**/*',r=>/analytics\.game100garage\.com|googlesyndication|google-analytics|googletagmanager/.test(r.request().url())?r.abort():r.continue());await page.addInitScript(()=>{window.__inputTrace=[];for(const name of ['pointerdown','pointerup','click','keydown','keyup'])document.addEventListener(name,e=>{window.__inputTrace.push({type:name,target:e.target?.id,key:e.key,state:document.querySelector('#app')?.dataset.state});if(window.__inputTrace.length>100)window.__inputTrace.shift();},true);});await page.goto(base+'/game027.html');await page.locator('#analytics-settings-host').locator('#deny').click();await page.locator('#mode-choice').selectOption('two');await page.locator('#play-button').click();await page.screenshot({path:path.join(dir,'normal-start.png')});
 for(let n=0;n<60;n++){const s=(await read()).snapshot;if(!s)break;const state=await page.locator('#app').getAttribute('data-state');let selected;
  if(state==='placing'){selected=planShot(s,true);const p=selected?.start||{x:225,y:225};const box=await page.locator('#table').boundingBox();await page.mouse.move(box.x+(p.x+50)/1000*box.width,box.y+(p.y+50)/550*box.height);await page.mouse.down();await page.mouse.up();await page.locator('#place-button').click();}
  const now=(await read()).snapshot;if(now.match.break)selected={angle:0,power:95,ball:1,strategy:'break'};else if(!selected)selected=planShot(now,false);

  await aim(selected.angle);await power(selected.power);await page.locator('#shoot-button').click();await page.waitForFunction(()=>document.querySelector('#app').dataset.state!=='moving',null,{timeout:25000});const after=await read();shots.push({number:n+1,shooter:now.match.turn,group:now.match.groups[now.match.turn],target:selected.ball,power:selected.power,strategy:selected.strategy,shotBefore:now,shotAfter:after.snapshot,stats:after.stats,remaining:after.snapshot?.world.balls.filter(b=>!b.pocketed).map(b=>b.id),state:await page.locator('#app').getAttribute('data-state')});fs.writeFileSync(path.join(dir,'progress.json'),JSON.stringify({runId,hash,shots},null,2));if(!after.snapshot){await page.screenshot({path:path.join(dir,'normal-result.png')});break;}
 }
 if(await page.locator('#app').getAttribute('data-state')!=='result')throw Error('bounded60shots did not reach normal result');const result=await read();fs.writeFileSync(path.join(dir,'normal-result-dom.html'),await page.content());if(result.stats.games!==1)throw Error('normal outcome must count once');await page.locator('#retry-button').click();await page.screenshot({path:path.join(dir,'normal-result-retry.png')});if((await read()).stats.games!==1)throw Error('retry changed match outcome stats');await page.locator('#pause-button').click();await page.locator('#title-button').click();await page.reload();if((await read()).stats.games!==1)throw Error('reload double-counted result');await page.locator('#play-button').click();await page.screenshot({path:path.join(dir,'normal-fresh-after-result.png')});await page.locator('#pause-button').click();await page.locator('#restart-button').click();await page.locator('#confirm-restart').click();await page.screenshot({path:path.join(dir,'normal-retry.png')});if((await read()).stats.games!==1)throw Error('restart changed match outcome stats');if(errors.length)throw Error('page errors');fs.writeFileSync(path.join(dir,'report.json'),JSON.stringify({runId,hash,collectorHash,plannerHash,base,pass:true,shots,resultStats:result.stats,errors,provenance:'normal compiled two-player match, read-only board assistance for geometric ordinary pointer aim/power/placement; no world, rules, outcome or save injection; human skill/fun not claimed'},null,2));console.log(JSON.stringify({dir,shots:shots.length,pass:true}));
}catch(e){if(page){try{await page.screenshot({path:path.join(dir,'FAILED.png')});fs.writeFileSync(path.join(dir,'FAILED-DOM.html'),await page.content());fs.writeFileSync(path.join(dir,'FAILED-FULL-STATE.json'),JSON.stringify(await read(),null,2));fs.writeFileSync(path.join(dir,'FAILED-TABLE.json'),JSON.stringify(await page.locator('#table').boundingBox()));trace=await page.evaluate(()=>window.__inputTrace);}catch{}}fs.writeFileSync(path.join(dir,'report.json'),JSON.stringify({runId,hash,collectorHash,plannerHash,base,pass:false,error:e.message,shots,errors,trace,provenance:'bounded normal-match native operation'},null,2));throw e;}finally{await browser.close();}
