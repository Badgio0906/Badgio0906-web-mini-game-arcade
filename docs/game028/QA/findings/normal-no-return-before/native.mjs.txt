import{chromium}from'playwright';import fs from'node:fs/promises';import path from'node:path';
const base=process.env.GAME028_BASE_URL??'http://127.0.0.1:4928';const out=process.env.GAME028_REPORT_DIR;if(!out)throw Error('GAME028_REPORT_DIR required: immutable unique directory');await fs.mkdir(out,{recursive:false});
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});const report={started_at:new Date().toISOString(),base,provenance:'compiled native normal input; readonly storage inspection; no physics injection; automated not human',runs:[]};
const key='web-mini-arcade:v1:game028:state';
try{for(const [name,width,height,touch]of[['desktop',1365,1000,false],['phone',390,844,true],['small',320,740,true],['landscape',844,390,true]]){
 const ctx=await browser.newContext({viewport:{width,height},hasTouch:touch,isMobile:touch});const page=await ctx.newPage();const errors=[],events=[];page.on('pageerror',e=>errors.push(e.message));await page.route(/analytics\.game100garage\.com|google-analytics\.com|googlesyndication\.com|doubleclick\.net/,r=>r.fulfill({status:204,body:''}));

 const phase=()=>page.locator('#app').getAttribute('data-state');const save=()=>page.evaluate(k=>JSON.parse(localStorage.getItem(k)??'null'),key);
 const check=(condition,label)=>{if(!condition)throw Error(`${name}: ${label}`);events.push(label);};
 await click('#explain');check(await phase()==='explanation','optional explanation');await page.screenshot({path:path.join(out,`${name}-explanation.png`)});await click('#practice');check(await phase()==='practice','score-free practice');await click('#serve');await page.waitForTimeout(5500);await click('#pause');check(await phase()==='paused','pause practice');await page.screenshot({path:path.join(out,`${name}-practice-pause.png`)});const beforePractice=await save();check(!beforePractice?.active,'practice excluded from saved RUN');check(beforePractice?.stats?.bestRally===0,'practice excluded from BEST');await click('#resume');check(await phase()==='practice','explicit practice resume');await click('#pause');await click('#back-title');

 const boundary=await page.evaluate(()=>{const c=document.querySelector('#board').getBoundingClientRect();return{x:c.x,y:c.y,w:c.width,h:c.height,vw:innerWidth,vh:innerHeight,bodyW:document.body.scrollWidth,bodyH:document.body.scrollHeight,buttons:[...document.querySelectorAll('button')].filter(b=>b.getClientRects().length&&!b.closest('dialog:not([open])')).map(b=>({id:b.id,w:b.getBoundingClientRect().width,h:b.getBoundingClientRect().height}))};});check(boundary.x>=0&&boundary.y>=0&&boundary.x+boundary.w<=width+.5&&boundary.y+boundary.h<=height+.5,'entire table/ball playfield fits viewport');check(boundary.bodyW<=width+.5&&boundary.bodyH<=height+.5,'no real-time page scroll');check(boundary.buttons.every(b=>b.w>=44&&b.h>=44),'44px controls');
 await click('#pause');const paused=await save();await page.reload({waitUntil:'networkidle'});await page.waitForTimeout(220);check(await phase()==='paused','reload restores paused');const reloaded=await save();check(JSON.stringify(paused.match)===JSON.stringify(reloaded.match),'restore no simulation catchup');check(paused.localReportId===reloaded.localReportId&&paused.observerRunId===reloaded.observerRunId,'local and existing observer IDs stable');await click('#resume');await click('#restart');check(await phase()==='confirm','restart explicit confirmation');await click('#cancel');check(await phase()==='playing','restart cancellation resumes');await click('#restart');await click('#confirm');const retrySave=await save();check(retrySave.localReportId!==paused.localReportId,'retry invalidates old match');check(retrySave.match.scores.join(',')==='0,0','retry resets scores');
 const box=await page.locator('#board').boundingBox();
 const move=async fraction=>{if(touch){const cdp=await ctx.newCDPSession(page);await cdp.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x:box.x+box.width*fraction,y:box.y+box.height*.87}]});await cdp.send('Input.dispatchTouchEvent',{type:'touchEnd',touchPoints:[]});await cdp.detach();}else await page.mouse.move(box.x+box.width*fraction,box.y+box.height*.87);};
 await move(.18);await click('#serve');await move(.82);for(let n=0;n<18&&await phase()!=='result';n++){await page.waitForTimeout(2600);if(await phase()==='result')break;const enabled=await page.locator('#serve').isEnabled();if(enabled){await move(.18);await click('#serve');await move(.82);}else await move(.12);}

 }}catch(error){report.failure={message:error.message,stack:error.stack};throw error;}finally{report.finished_at=new Date().toISOString();await fs.writeFile(path.join(out,'report.json'),JSON.stringify(report,null,2));await browser.close();}
